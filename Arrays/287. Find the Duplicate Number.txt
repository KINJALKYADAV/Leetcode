Brute

class Solution {
public:
    int findDuplicate(vector<int>& nums) {
        sort(nums.begin(),nums.end());
      for(int i=0;i<nums.size()-1;i++)
      {
        if(nums[i]==nums[i+1])
            return nums[i];
      }  
      return 0;
    }
};

Better

class Solution {
public:
    int findDuplicate(vector<int>& nums) {
        int n = nums.size();
        vector<int> freq(n,0);
        for(int i =0;i<n;i++)
        {
            if(freq[nums[i]]==1)
            {
                return nums[i];
            }
            else
                freq[nums[i]]++;
        }  
      return 0;
    }
};


Optimal

class Solution {
public:
    int findDuplicate(vector<int>& nums) {
        int n = nums.size();
        int slow = nums[0];
        int fast = nums[0];
        do
        {
            slow = nums[slow];
            fast = nums[nums[fast]];
        }while(slow!=fast);
        fast = nums[0];
        while(slow!= fast)
        {
            slow= nums[slow];
            fast = nums[fast];
        }
      return slow;
    }
};
